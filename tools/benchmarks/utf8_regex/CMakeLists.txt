cmake_minimum_required(VERSION 3.30)
project(utf8_regex_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(PCRE2_QT_SOURCE_DIR "" CACHE PATH "Explicit Qt bundled PCRE2 10.47 source directory; never downloaded")
option(UTF8_REGEX_SANITIZE "Enable address/undefined sanitizers for the explicit tool" OFF)
if(NOT EXISTS "${PCRE2_QT_SOURCE_DIR}/src/config.h" OR NOT EXISTS "${MIMALLOC_LIBRARY}")
  message(FATAL_ERROR "Set PCRE2_QT_SOURCE_DIR and the project's built MIMALLOC_LIBRARY")
endif()
include(cmake/Pcre2.cmake)
zzlogg_add_probe_pcre2(probe_pcre2_8 "${PCRE2_QT_SOURCE_DIR}")
find_package(Qt6 6.10 REQUIRED COMPONENTS Core)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(utf8_regex_probe main.cpp matcher.cpp contract.cpp)
target_include_directories(utf8_regex_probe PRIVATE
  "${REPO_ROOT}/src/regex/include" "${REPO_ROOT}/src/utils/include"
  "${REPO_ROOT}/3rdparty/vendor/mimalloc/include"
  "${REPO_ROOT}/3rdparty/vendor/type_safe/include"
  "${REPO_ROOT}/3rdparty/vendor/type_safe/external/debug_assert")
target_compile_definitions(utf8_regex_probe PRIVATE TYPE_SAFE_ENABLE_ASSERTIONS=0)
target_link_libraries(utf8_regex_probe PRIVATE probe_pcre2_8 Qt6::Core "${MIMALLOC_LIBRARY}")
if(MSVC)
  target_compile_options(utf8_regex_probe PRIVATE /utf-8)
endif()
if(UTF8_REGEX_SANITIZE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizer option needs GCC or Clang")
  endif()
  foreach(target probe_pcre2_8 utf8_regex_probe)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
  target_link_options(utf8_regex_probe PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME utf8_regex_contract COMMAND utf8_regex_probe --verify)
