# Recompile the production translation units; never mutate the production targets.
function(worker_probe_copy original copy)
  get_target_property(_sources ${original} SOURCES)
  add_library(${copy} STATIC EXCLUDE_FROM_ALL ${_sources})
  foreach(_property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS AUTOMOC INTERPROCEDURAL_OPTIMIZATION)
    get_target_property(_value ${original} ${_property})
    if(NOT _value STREQUAL "_value-NOTFOUND")
      set_property(TARGET ${copy} PROPERTY ${_property} "${_value}")
    endif()
  endforeach()
  get_target_property(_definitions ${original} INTERFACE_COMPILE_DEFINITIONS)
  if(NOT _definitions STREQUAL "_definitions-NOTFOUND")
    target_compile_definitions(${copy} PUBLIC ${_definitions})
  endif()
  get_target_property(_includes ${original} INTERFACE_INCLUDE_DIRECTORIES)
  target_include_directories(${copy} PUBLIC ${_includes} ${CMAKE_CURRENT_SOURCE_DIR})
  get_target_property(_links ${original} LINK_LIBRARIES)
  list(TRANSFORM _links REPLACE "^klogg_regex$" "zzlogg_probe_regex")
  target_link_libraries(${copy} PUBLIC ${_links})
  target_compile_definitions(${copy} PUBLIC ZZLOGG_EXPERIMENTAL_SEARCH)
endfunction()
worker_probe_copy(klogg_regex zzlogg_probe_regex)
target_sources(zzlogg_probe_regex PRIVATE experimental.cpp ../literal_search/matcher.cpp)
target_include_directories(zzlogg_probe_regex PRIVATE ../literal_search)
target_compile_definitions(zzlogg_probe_regex PRIVATE LITERAL_ENABLE_RUST)
target_link_libraries(zzlogg_probe_regex PRIVATE zzlogg_literal_probe_rust)
worker_probe_copy(klogg_logdata zzlogg_probe_logdata)
